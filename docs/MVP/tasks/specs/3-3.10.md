### Derived Field Calculations
```python
def compute_engagement_rate(item: MediaItemInput) -> float | None:
    """
    engagement_rate = (likes + comments + shares) / plays
    Capped at 1.0 for viral content
    """
    if not item.play_count or item.play_count == 0:
        return None

    engagements = (
        (item.like_count or 0) +
        (item.comment_count or 0) +
        (item.share_count or 0)
    )
    rate = engagements / item.play_count
    return min(rate, 1.0)

def infer_user_role(item: MediaItemInput, hour: int) -> str:
    """
    Heuristic-based user role inference:
    - wind_down: late night (22-4) + calm/funny content
    - learning: tutorial content + favorited
    - entertainment: comedy/entertainment content
    - inspiration: motivational/inspirational content
    - discovery: diverse categories
    """
    if hour >= 22 or hour <= 4:
        if item.mood_primary in ['calm_soothing', 'funny']:
            return 'wind_down'

    if item.content_category_primary == 'tutorial':
        return 'learning'

    if item.mood_primary == 'inspirational':
        return 'inspiration'

    if item.content_category_primary == 'entertainment':
        return 'entertainment'

    return 'discovery'
```


### creator_is_repeat query
```sql
-- Check if user has other media_events from same creator
SELECT EXISTS (
    SELECT 1 FROM media_events
    WHERE user_id = $user_id
      AND creator_username = $creator_username
      AND id != $media_event_id
) AS is_repeat;
```

### processing_steps insert
```sql
INSERT INTO processing_steps (
```
