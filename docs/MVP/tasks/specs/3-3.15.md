### Progress Update Function
```python
from datetime import datetime, timedelta
from uuid import UUID

async def update_progress(
    upload_id: UUID,
    step_name: str,
    items_processed: int,
    items_failed: int = 0,
    cost_usd: float = 0.0,
    db: AsyncSession = None,
) -> None:
    """
    Atomically update progress counters for a pipeline step.
    Uses SELECT FOR UPDATE to prevent race conditions.
    """
    counter_column = STEP_TO_COUNTER[step_name]

    await db.execute(
        text("""
            UPDATE upload_pipeline_runs SET
                {counter_column} = {counter_column} + :items_processed,
                items_failed = items_failed + :items_failed,
                total_cost_usd = total_cost_usd + :cost_usd,
                current_step = :step_name,
                estimated_completion_at = :estimated_at,
                updated_at = NOW()
            WHERE upload_id = :upload_id
        """.format(counter_column=counter_column)),
        {
            "items_processed": items_processed,
            "items_failed": items_failed,
            "cost_usd": cost_usd,
            "step_name": step_name,
            "estimated_at": calculate_estimated_completion(upload_id, step_name),
            "upload_id": str(upload_id),
        }
    )
    await db.commit()


STEP_TO_COUNTER = {
    "PARSE_EXPORT": None,  # Sets total_items, not a counter
    "APIFY_ENRICH": "items_enriched",
    "MEDIA_DOWNLOAD": "items_media_downloaded",
    "SUBTITLE_FETCH": None,  # Included in transcribed
    "WHISPER_TRANSCRIBE": "items_transcribed",
    "VISION_ANALYSIS": "items_vision_done",
    "TEXT_FUSION": None,  # Included in embedded
    "EMBEDDING": "items_embedded",
    "DERIVED_FIELDS": None,  # Included in complete
    "SEARCH_INDEX": "items_complete",
}
```

### Estimated Completion Calculation
```python
def calculate_estimated_completion(
    upload_id: UUID,
    current_step: str,
    items_processed_in_step: int,
    step_start_time: datetime,
) -> datetime:
    """
    Estimate completion time based on current processing rate.
    """
    # Get current progress
    run = get_pipeline_run(upload_id)
    total_items = run.total_items

    # Calculate rate from current step
    elapsed = (datetime.utcnow() - step_start_time).total_seconds()
    if elapsed > 0 and items_processed_in_step > 0:
        items_per_second = items_processed_in_step / elapsed
    else:
        items_per_second = 1.0  # Default fallback

    # Estimate remaining items across all steps
    remaining_steps = STEPS_AFTER[current_step]
    remaining_items = sum(
        total_items - getattr(run, STEP_TO_COUNTER.get(step, 'items_complete'))
        for step in remaining_steps
        if STEP_TO_COUNTER.get(step)
    )

    # Calculate estimated time
    estimated_seconds = remaining_items / items_per_second
    # Apply step-specific multipliers (some steps are slower)
    estimated_seconds *= STEP_SPEED_MULTIPLIER.get(current_step, 1.0)

    return datetime.utcnow() + timedelta(seconds=estimated_seconds)
```
