### Text Fusion Algorithm
```python
def fuse_text(item: MediaItemInput) -> str:
    """
    Combine text sources with weighting:
    1. Caption (primary, full weight)
    2. Hashtags (converted to readable text)
    3. Transcript/subtitles (full weight)
    4. OCR text (lower weight - may have noise)
    5. Visual tags (context, lower weight)
    6. Apparent intent (summary)
    """
    parts = []

    if item.caption_text:
        parts.append(item.caption_text)

    if item.hashtags:
        # Convert hashtags to readable text
        parts.append(" ".join(item.hashtags))

    if item.subtitle_text:
        parts.append(item.subtitle_text)

    if item.ocr_text:
        # OCR may duplicate caption, deduplicate
        ocr_clean = deduplicate_from_caption(item.ocr_text, item.caption_text)
        if ocr_clean:
            parts.append(ocr_clean)

    if item.visual_tags:
        parts.append(" ".join(item.visual_tags))

    if item.apparent_intent:
        parts.append(f"Intent: {item.apparent_intent}")

    full_text = " ".join(parts)
    return normalize_whitespace(full_text)
```
