// src/frontend/components/upload/TikTokUploader.tsx
import { z } from 'zod';

export const uploadResultSchema = z.object({
  uploadId: z.string().uuid(),
  storagePath: z.string(),
  filename: z.string(),
});

export type UploadResult = z.infer<typeof uploadResultSchema>;

export interface TikTokUploaderProps {
  /** Called when upload completes successfully */
  onUploadComplete: (result: UploadResult) => void;
  /** Called when upload fails */
  onUploadError?: (error: Error) => void;
  /** Called when upload is cancelled */
  onUploadCancel?: () => void;
  /** Maximum file size in bytes (default: 500MB) */
  maxFileSize?: number;
  /** Whether the uploader is disabled */
  disabled?: boolean;
  /** Custom class name for styling */
  className?: string;
}

export interface UploadState {
  status: 'idle' | 'uploading' | 'complete' | 'error';
  progress: number; // 0-100
  bytesUploaded: number;
  bytesTotal: number;
  filename: string | null;
  error: Error | null;
}

### Uppy Configuration

```typescript
// src/frontend/lib/uppy/config.ts
import Uppy from '@uppy/core';
import XHRUpload from '@uppy/xhr-upload';

export const ALLOWED_FILE_TYPES = [
  'application/zip',
  'application/x-zip-compressed',
  'application/octet-stream',
];

export const MAX_FILE_SIZE = 500 * 1024 * 1024; // 500MB

export function createUppyInstance() {
  return new Uppy({
    restrictions: {
      maxFileSize: MAX_FILE_SIZE,
      maxNumberOfFiles: 1,
      allowedFileTypes: ALLOWED_FILE_TYPES,
    },
    autoProceed: false,
    allowMultipleUploadBatches: false,
  });
}
```

## 5) Workflow & state machine

### Upload States
- `idle` - No file selected, waiting for user
- `uploading` - File upload in progress
- `complete` - Upload finished successfully
- `error` - Upload failed

### Transitions
- `idle` → `uploading`: User selects file and clicks upload
- `uploading` → `complete`: XHR upload succeeds
- `uploading` → `error`: XHR upload fails
- `uploading` → `idle`: User cancels upload
- `error` → `idle`: User clicks retry or selects new file
- `complete` → `idle`: Component reset for new upload

### Upload Flow
1. User drags or selects ZIP file
2. Client validates file type and size
3. Client calls `POST /api/uploads/presigned-url` to get upload URL
4. Uppy uploads file directly to Supabase Storage via XHR
5. On success, callback with upload_id and storage_path
6. Parent component handles next steps (scope selection, etc.)
